# STEP 1: Check Docker image
docker images

# STEP 2: Check running containers
docker ps

# STEP 3: Check Kubernetes version
kubectl version

# STEP 4: Check Kubernetes nodes
kubectl get nodes

# STEP 5: Check active Kubernetes context
kubectl config current-context

# STEP 6: Check Docker image port
docker image inspect firstapp:latest --format '{{json .Config.ExposedPorts}}'

# STEP 7: Deploy Docker image in Kubernetes
kubectl create deployment firstapp --image=firstapp:latest --image-pull-policy=IfNotPresent

# STEP 8: Check deployment
kubectl get deployments

# STEP 9: Check Pods
kubectl get pods

# STEP 10: Describe deployment
kubectl describe deployment firstapp

# STEP 11: Check application logs
kubectl logs deployment/firstapp

# STEP 12: Expose application (assuming port 3000)
kubectl expose deployment firstapp --type=NodePort --port=3000 --target-port=3000

# STEP 13: Check services
kubectl get services

# STEP 14: Forward Kubernetes service to localhost
kubectl port-forward service/firstapp 8080:3000
