========================
QUESTION 1 – DEPARTMENT WEB
========================

mkdir department-web
cd department-web

notepad index.html

notepad Dockerfile

docker build -t departmentweb .

docker run -d --name department-site -p 8080:80 departmentweb

docker ps

http://localhost:8080

docker logs department-site

docker tag departmentweb adarshr2028/departmentweb:latest

docker login

docker push adarshr2028/departmentweb:latest

docker pull adarshr2028/departmentweb:latest

docker run -d --name department-site-test -p 8081:80 adarshr2028/departmentweb:latest

http://localhost:8081


========================
QUESTION 2 – STUDENT HELP DESK
========================

mkdir student-helpdesk
cd student-helpdesk

notepad index.html

notepad Dockerfile

docker build -t studenthelpdesk .

docker run -d --name helpdesk -p 8088:80 studenthelpdesk

docker ps

docker exec helpdesk ls /usr/share/nginx/html/index.html

docker logs helpdesk

docker stop helpdesk

docker start helpdesk

docker tag studenthelpdesk YOUR_USERNAME/studenthelpdesk:latest

docker login

docker push YOUR_USERNAME/studenthelpdesk:latest

docker pull YOUR_USERNAME/studenthelpdesk:latest

docker run -d --name helpdesk-test -p 8089:80 YOUR_USERNAME/studenthelpdesk:latest

http://localhost:8089
```

### Dockerfile for both

```dockerfile
FROM nginx:latest
COPY index.html /usr/share/nginx/html/index.html
EXPOSE 80
