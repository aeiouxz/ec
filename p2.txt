2. Product Sales Exploratory Analysis
import pandas as pd
import matplotlib.pyplot as plt

df = pd.read_csv("productsales.csv")

print("Shape of Dataset:")
print(df.shape)

print("\nData Types:")
print(df.dtypes)

print("\nStatistical Summary:")
print(df.describe())

category_count = df["Category"].value_counts()

print("\nNumber of Products in Each Category:")
print(category_count)

print("\nCategory with Highest Number of Products:")
print(category_count.idxmax())

print("\nCategory with Lowest Number of Products:")
print(category_count.idxmin())

print("\nCorrelation between Units Sold and Profit:")
print(df["Unit_Sold"].corr(df["Profit"]))

plt.figure(figsize=(6, 4))
plt.scatter(df["Unit_Sold"], df["Profit"])
plt.xlabel("Units Sold")
plt.ylabel("Profit")
plt.title("Units Sold vs Profit")
plt.show()

plt.figure(figsize=(6, 4))
plt.hist(df["Selling_Price"], bins=5)
plt.xlabel("Selling Price")
plt.ylabel("Frequency")
plt.title("Distribution of Selling Price")
plt.show()

plt.figure(figsize=(6, 4))
plt.boxplot(df["Selling_Price"])
plt.ylabel("Selling Price")
plt.title("Box Plot of Selling Price")
plt.show()
