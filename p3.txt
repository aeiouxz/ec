# STEP 1: Install CPU Checker
sudo apt update
sudo apt install -y cpu-checker

# STEP 2: Check Virtualization Support
sudo kvm-ok

# STEP 3: Install KVM and Required Packages
sudo apt install -y qemu-kvm libvirt-daemon libvirt-clients bridge-utils virt-manager

# STEP 4: Check libvirtd Service Status
sudo systemctl status libvirtd

# STEP 5: Enable libvirtd to Start on Boot
sudo systemctl enable libvirtd

# STEP 6: Start libvirtd Service
sudo systemctl start libvirtd

# STEP 7: Verify KVM Installation
sudo virsh list --all

# STEP 8: Create Windows 10 Virtual Machine Using virt-install
sudo virt-install \
  --name=win10-vm \
  --os-variant=win10 \
  --vcpus=2 \
  --ram=4096 \
  --disk size=64 \
  --graphics spice \
  --cdrom=/home/mca/Documents/Win10_2004_English_x64.iso \
  --network network=default

# STEP 9: Open Virtual Machine Manager
virt-manager
