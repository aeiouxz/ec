import numpy as np
import matplotlib.pyplot as plt

production = np.array([95, 98, 100, 97, 102, 99, 101, 96, 103, 250])

mean = np.mean(production)

median = np.median(production)

variance = np.var(production)

std = np.std(production)

print("Mean:", mean)
print("Median:", median)
print("Variance:", variance)
print("Standard Deviation:", std)

plt.figure(figsize=(6, 4))
plt.boxplot(production)
plt.ylabel("Production")
plt.title("Box Plot of Daily Production")
plt.show()

plt.figure(figsize=(6, 4))
plt.hist(production, bins=5)
plt.xlabel("Production")
plt.ylabel("Frequency")
plt.title("Distribution of Daily Production")
plt.show()

production_without_outlier = production[production != 250]

print("\nStatistics without Outlier:")
print("Mean:", np.mean(production_without_outlier))
print("Standard Deviation:", np.std(production_without_outlier))

print("\nConclusion:")
print("250 is an outlier.")
print("Median is a more appropriate measure of typical production.")
