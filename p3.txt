# 1. Install CPU checker
sudo apt install -y cpu-checker

# 2. Check KVM support
kvm-ok

# 3. Install KVM packages
sudo apt install -y qemu-kvm libvirt-daemon libvirt-clients bridge-utils virt-manager

# 4. Check libvirt service
sudo systemctl status libvirtd

# 5. Start and enable libvirt
sudo systemctl enable --now libvirtd

# 6. Check Windows ISO
ls -l /home/mca/Documents/Win10_2004_English_x64.iso

# 7. Create Windows 10 VM
sudo virt-install \
  --name=win10-vm \
  --os-variant=win10 \
  --vcpus=2 \
  --ram=4096 \
  --disk size=64 \
  --graphics spice \
  --cdrom=/home/mca/Documents/Win10_2004_English_x64.iso \
  --network network=default

# 8. Install Windows 10
# Use the Windows installer inside the VM

# 9. Check all VMs
virsh list --all

# 10. Start VM
virsh start win10-vm

# 11. Shut down VM
virsh shutdown win10-vm

# 12. Force stop VM
virsh destroy win10-vm
