Create an Android application to perform the four basic arithmetic operations using the following widgets.

EditText

Number 1
Number 2

Buttons

Calculate
Reset

TextViews

Addition Result
Subtraction Result
Multiplication Result
Division Result

When Calculate is clicked, display all four results. When Reset is clicked, clear both input fields and all four results.


MainActivity.java
package com.example.adarsh2;

import android.os.Bundle;
import android.view.View;
import android.widget.Button;
import android.widget.EditText;
import android.widget.TextView;

import androidx.activity.EdgeToEdge;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {

    EditText et1, et2;
    Button btnCalculate, btnReset;
    TextView tvAdd, tvSub, tvMul, tvDiv;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_main);

        et1 = findViewById(R.id.et1);
        et2 = findViewById(R.id.et2);

        btnCalculate = findViewById(R.id.btnCalculate);
        btnReset = findViewById(R.id.btnReset);

        tvAdd = findViewById(R.id.tvAdd);
        tvSub = findViewById(R.id.tvSub);
        tvMul = findViewById(R.id.tvMul);
        tvDiv = findViewById(R.id.tvDiv);

        // Calculate button
        btnCalculate.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View v) {

                // Get numbers from EditText
                double num1 =
                        Double.parseDouble(et1.getText().toString());

                double num2 =
                        Double.parseDouble(et2.getText().toString());

                // Display addition
                tvAdd.setText("Addition: " + (num1 + num2));

                // Display subtraction
                tvSub.setText("Subtraction: " + (num1 - num2));

                // Display multiplication
                tvMul.setText("Multiplication: " + (num1 * num2));

                // Check before division
                if (num2 != 0) {
                    tvDiv.setText("Division: " + (num1 / num2));
                } else {
                    tvDiv.setText("Division: Cannot divide by zero");
                }
            }
        });

        // Reset button
        btnReset.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View v) {

                // Clear input fields
                et1.setText("");
                et2.setText("");

                // Clear results
                tvAdd.setText("");
                tvSub.setText("");
                tvMul.setText("");
                tvDiv.setText("");
            }
        });
    }
}
