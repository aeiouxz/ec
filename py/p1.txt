# 09 - ATM Menu
# ============================================================
balance = 1000.0

while True:
    print("\nATM MENU")
    print("1. Deposit")
    print("2. Withdraw")
    print("3. Balance Enquiry")
    print("4. Exit")

    choice = int(input("Enter choice: "))

    if choice == 1:
        amount = float(input("Enter deposit amount: "))
        if amount > 0:
            balance += amount
            print("Deposited:", amount)
            print("Balance:", balance)
        else:
            print("Invalid amount")

    elif choice == 2:
        amount = float(input("Enter withdrawal amount: "))
        if amount <= 0:
            print("Invalid amount")
        elif amount > balance:
            print("Insufficient balance")
        else:
            balance -= amount
            print("Withdrawn:", amount)
            print("Balance:", balance)

    elif choice == 3:
        print("Available balance:", balance)

    elif choice == 4:
        print("Exiting...")
        break

    else:
        print("Invalid choice")

# 14 - Prime Number Check
# ============================================================
def is_prime(n):
    if n < 2:
        return False

    for i in range(2, int(n ** 0.5) + 1):
        if n % i == 0:
            return False

    return True

n = int(input("Enter number: "))

if is_prime(n):
    print("Prime Number")
else:
    print("Not a Prime Number")

n = int(input("Enter number of entries: "))

for i in range(n):
    key = input("Enter key: ")
    value = input("Enter value: ")
    student[key] = value

print("Original dictionary:", student)

update_key = input("Enter key to update: ")
update_value = input("Enter new value: ")
student[update_key] = update_value

delete_key = input("Enter key to delete: ")

if delete_key in student:
    del student[delete_key]
else:
    print("Key not found")

print("Updated dictionary:", student)

# 26 - Probability of Even Dice Number
# ============================================================
import random

rolls = [random.randint(1, 6) for _ in range(100)]

# 31 - Search Student by Register Number

import csv

register_number = input("Enter register number: ")

with open("students.csv", "r") as file:
    reader = csv.DictReader(file)
    found = False

    for student in reader:
        if student["Register Number"] == register_number:
            print("Student Found")
            print("Register Number:", student["Register Number"])
            print("Name:", student["Name"])
            print("Marks:", student["Marks"])
            found = True
            break

if not found:
    print("Student Not Found")
